## 字符串反转

三种写法，从 API 组合到递归：

```js
// Array.reverse()
function reverseString(str) {
  return str.split('').reverse().join('');
}

// for 循环倒序拼接
function reverseString(str) {
  let reversedStr = '';
  for (let i = str.length - 1; i >= 0; i--) {
    reversedStr += str[i];
  }
  return reversedStr;
}

// 递归：每次取第一个字符放到末尾
function reverseString(str) {
  if (str === '') {
    return '';
  } else {
    return reverseString(str.substr(1)) + str.charAt(0);
  }
}
```

## 数组扁平化

四种实现，面试常要求手写后三种：

```js
// 1. 原生 flat
arr.flat(Infinity);

// 2. 扩展运算符：只要还有嵌套就拍一层
function flatten1(arr) {
  while (arr.some((item) => Array.isArray(item))) {
    arr = [].concat(...arr);
  }
  return arr;
}

// 3. reduce 递归
function flatten2(arr) {
  return arr.reduce((prev, cur) => {
    if (Array.isArray(cur)) {
      return prev.concat(flatten2(cur));
    } else {
      return prev.concat(cur);
    }
  }, []);
}

// 4. 基础递归
function flatten3(arr) {
  let res = [];
  for (let i = 0; i < arr.length; i++) {
    if (Array.isArray(arr[i])) {
      res = res.concat(flatten3(arr[i]));
    } else {
      res.push(arr[i]);
    }
  }
  return res;
}
```

## 数组去重

四种方式，追问点通常是各自的适用场景（NaN、对象引用、性能）：

```js
// Set：最简洁，NaN 也能正确去重
const newArr = [...new Set(arr)];

// for + indexOf： indexOf 找到的是第一个位置，等于当前 index 说明首次出现
function unique(arr) {
  const newArr = [];
  for (let i = 0; i < arr.length; i++) {
    if (arr.indexOf(arr[i]) === i) {
      newArr.push(arr[i]);
    }
  }
  return newArr;
}

// for + includes：结果顺序稳定
function unique(arr) {
  const newArr = [];
  for (let i = 0; i < arr.length; i++) {
    if (!newArr.includes(arr[i])) {
      newArr.push(arr[i]);
    }
  }
  return newArr;
}

// filter 写法
function unique(arr) {
  return arr.filter((item, index) => arr.indexOf(item) === index);
}
```

注意 `indexOf` 内部用严格相等（`===`），所以 `NaN` 永远查不到自身（`[NaN].indexOf(NaN) === -1`），只有 Set 方案能对 NaN 去重。

## 设计题：按 id 批量取数，优先走缓存

> 写一个通用工具类，通过 id 集合获取数据集合；先查缓存，未命中再调服务查询并写入缓存，最后返回数据集合。

考察点：缓存数据结构选型（Map）、批量流程编排、边界处理：

```js
const api = {
  getDataById: (id) => {
    return 'dataFromService';
  },
};

class DataFetch {
  constructor() {
    this.cache = new Map();
  }

  getDataList(idList) {
    const dataList = [];
    idList.forEach((id) => {
      if (this.cache.has(id)) {
        dataList.push(this.cache.get(id));
      } else {
        const data = api.getDataById(id);
        if (data) {
          this.cache.set(id, data);
          dataList.push(data);
        }
      }
    });
    return dataList; // 原笔记漏了这行 return
  }
}
```

可以延伸的方向：把同步的 `api.getDataById` 换成异步接口（`Promise.all` 并行请求未命中的 id）、加缓存容量上限（LRU）。
